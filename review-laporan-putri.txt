<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\TargetProduksi;
use Illuminate\Http\Request;

class LaporanProduksiController extends Controller
{
    public function index(Request $request)
    {
        $request->validate([
            'tanggal_mulai' => 'nullable|date',
            'tanggal_akhir' => 'nullable|date|after_or_equal:tanggal_mulai',
        ]);

        $query = TargetProduksi::query()
            ->with(['produk', 'penenun'])
            ->withSum('hasilProduksi as total_selesai', 'jumlah_selesai');

        if ($request->filled('tanggal_mulai')) {
            $query->whereDate(
                'tanggal_mulai',
                '>=',
                $request->tanggal_mulai
            );
        }

        if ($request->filled('tanggal_akhir')) {
            $query->whereDate(
                'tanggal_mulai',
                '<=',
                $request->tanggal_akhir
            );
        }

        // Ringkasan dihitung dari seluruh data yang difilter.
        $ringkasan = (clone $query)->get();

        $totalTarget = $ringkasan->sum('jumlah_target');

        $totalSelesai = $ringkasan->sum(
            fn ($item) => $item->total_selesai ?? 0
        );

        $persentase = $totalTarget > 0
            ? round(($totalSelesai / $totalTarget) * 100, 1)
            : 0;

        // Tabel ditampilkan per halaman.
        $laporan = $query
            ->orderByDesc('tanggal_mulai')
            ->orderByDesc('id')
            ->paginate(10)
            ->withQueryString();

        return view('admin.laporan.produksi', compact(
            'laporan',
            'totalTarget',
            'totalSelesai',
            'persentase'
        ));
    }
}

<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\Stok;
use App\Models\MutasiStok;
use Illuminate\Http\Request;

class LaporanStokController extends Controller
{
    public function index(Request $request)
    {
        $request->validate([
            'tanggal_mulai' => 'nullable|date',
            'tanggal_akhir' => 'nullable|date|after_or_equal:tanggal_mulai',
            'jenis_mutasi' => 'nullable|in:masuk,keluar,penyesuaian',
        ]);

        // Ringkasan stok saat ini
        $totalJenisProduk = Stok::whereHas('produk')->count();

        $totalStok = Stok::whereHas('produk')
            ->sum('jumlah_stok');

        $stokHabis = Stok::whereHas('produk')
            ->where('jumlah_stok', '<=', 0)
            ->count();

        // Daftar stok per produk
        $daftarStok = Stok::with('produk')
            ->whereHas('produk')
            ->orderBy('produk_id')
            ->paginate(10, ['*'], 'stok_page')
            ->withQueryString();

        // Riwayat mutasi stok
        $queryMutasi = MutasiStok::with('stok.produk');

        if ($request->filled('tanggal_mulai')) {
            $queryMutasi->whereDate(
                'tanggal_mutasi',
                '>=',
                $request->tanggal_mulai
            );
        }

        if ($request->filled('tanggal_akhir')) {
            $queryMutasi->whereDate(
                'tanggal_mutasi',
                '<=',
                $request->tanggal_akhir
            );
        }

        if ($request->filled('jenis_mutasi')) {
            $queryMutasi->where(
                'jenis_mutasi',
                $request->jenis_mutasi
            );
        }

        // Ringkasan mutasi sesuai filter
        $totalMasuk = (clone $queryMutasi)
            ->where('jenis_mutasi', 'masuk')
            ->sum('jumlah');

        $totalKeluar = (clone $queryMutasi)
            ->where('jenis_mutasi', 'keluar')
            ->sum('jumlah');

        $riwayatMutasi = $queryMutasi
            ->orderByDesc('tanggal_mutasi')
            ->orderByDesc('id')
            ->paginate(10, ['*'], 'mutasi_page')
            ->withQueryString();

        return view('admin.laporan.stok', compact(
            'totalJenisProduk',
            'totalStok',
            'stokHabis',
            'daftarStok',
            'totalMasuk',
            'totalKeluar',
            'riwayatMutasi'
        ));
    }
}

<?php
namespace App\Models;
use Illuminate\Database\Eloquent\Model;
class MutasiStok extends Model {
    protected $table='mutasi_stok';
    protected $fillable=['stok_id','jenis_mutasi','jumlah','sumber','hasil_produksi_id','detail_pesanan_id','catatan','tanggal_mutasi'];
    protected function casts(): array { return ['tanggal_mutasi'=>'datetime']; }
    public function stok(){ return $this->belongsTo(Stok::class,'stok_id'); }
    public function hasilProduksi(){ return $this->belongsTo(HasilProduksi::class,'hasil_produksi_id'); }
    public function detailPesanan(){ return $this->belongsTo(DetailPesanan::class,'detail_pesanan_id'); }
}
